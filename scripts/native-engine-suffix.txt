   append_(SHEETS.files,{object_id:p.object_id||uid_(),'Исходный путь':p.file_name||'','Размер, байт':p.file_size||0,'Создан':now_(),'Google Drive URL':path,'Статус миграции':'Приватное хранилище','Тип':p.slot==='signed'?'Подписанный документ':'Фото','Связанная приёмка':rec['Номер']||rec['Номер ремонта'],'Примечание':'FastGo',updated_at:now_(),record_id:p.id,slot:p.slot||'photos'});

 }
